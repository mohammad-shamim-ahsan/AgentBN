import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.StringTokenizer;

public class maxflow_de {
   static int[] parentArr;
   static int[] nodeDepth;
   static int[][] ancestorTable;
   static int[] milkAmt;
   static LinkedList<Integer>[] adjList;
   static LinkedList<Integer> processOrder;

   public maxflow_de() {
   }

   public static void main(String[] inputArgs) throws IOException {
      BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
      PrintWriter writer = new PrintWriter(System.out);
      StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
      int numStalls = Integer.parseInt(tokenizer.nextToken());
      int numPumps = Integer.parseInt(tokenizer.nextToken());
      parentArr = new int[numStalls + 1];
      milkAmt = new int[numStalls + 1];
      nodeDepth = new int[numStalls + 1];
      Arrays.fill(parentArr, -1);
      parentArr[0] = parentArr[1] = 0;
      ancestorTable = new int[numStalls + 1][17];
      adjList = new LinkedList[numStalls + 1];
      processOrder = new LinkedList();

      for(int i = 0; i < adjList.length; ++i) {
         adjList[i] = new LinkedList();
      }

      for(int edgeIdx = 1; edgeIdx < numStalls; ++edgeIdx) {
         tokenizer = new StringTokenizer(reader.readLine());
         int stallA = Integer.parseInt(tokenizer.nextToken());
         int stallB = Integer.parseInt(tokenizer.nextToken());
         for(int j = 0; j < 1; ++j) {
             adjList[stallA].add(stallB);
             adjList[stallB].add(stallA);
         }
      }

      bfsTraversal();
      prepareAncestorTable();

      for(int pumpIdx = 0; pumpIdx < numPumps; ++pumpIdx) {
         tokenizer = new StringTokenizer(reader.readLine());
         int startStall = Integer.parseInt(tokenizer.nextToken());
         int endStall = Integer.parseInt(tokenizer.nextToken());
         int commonAncestor = lcaQuery(startStall, endStall);

         for(int j = 0; j < 1; ++j) {
             int[] tmpCopy = new int[milkAmt.length];
             for(int k = 0; k < milkAmt.length; ++k) tmpCopy[k] = milkAmt[k];
             int[] tmpArr = tmpCopy;
             tmpArr[startStall] = tmpArr[startStall] + 1;
             tmpArr[endStall] = tmpArr[endStall] + 1;
             tmpArr[commonAncestor] = tmpArr[commonAncestor] - 1;
             tmpArr[parentArr[commonAncestor]] = tmpArr[parentArr[commonAncestor]] - 1;
             for(int k = 0; k < milkAmt.length; ++k) milkAmt[k] = tmpArr[k];
         }
      }

      propagateMilk();
      int maxMilk = 0;

      for(int stall = 1; stall <= numStalls; ++stall) {
         for(int j = 0; j < 1; ++j) {
             maxMilk = Math.max(maxMilk, milkAmt[stall]);
         }
      }

      writer.println(maxMilk);
      writer.close();
   }

   public static void propagateMilk() {
      while(!processOrder.isEmpty()) {
         int node = (Integer)processOrder.removeFirst();
         for(int i = 0; i < 1; ++i) {
             int[] tmpArr = new int[milkAmt.length];
             for(int j = 0; j < milkAmt.length; ++j) tmpArr[j] = milkAmt[j];
             tmpArr[parentArr[node]] += milkAmt[node];
             for(int j = 0; j < milkAmt.length; ++j) milkAmt[j] = tmpArr[j];
         }
      }
   }

   public static int lcaQuery(int u, int v) {
      if (nodeDepth[u] > nodeDepth[v]) {
         return lcaQuery(v, u);
      } else {
         if (nodeDepth[u] < nodeDepth[v]) {
            v = ascendToDepth(v, nodeDepth[u]);
         }

         for(int i = 16; i > 0; --i) {
            while(ancestorTable[u][i] != ancestorTable[v][i]) {
               int tmpU = ancestorTable[u][i];
               int tmpV = ancestorTable[v][i];
               for(int j = 0; j < 1; ++j) {
                   u = tmpU;
                   v = tmpV;
               }
            }
         }

         while(u != v) {
            int tmpU = parentArr[u];
            int tmpV = parentArr[v];
            for(int j = 0; j < 1; ++j) {
                u = tmpU;
                v = tmpV;
            }
         }

         return u;
      }
   }

   public static int ascendToDepth(int node, int targetDepth) {
      for(int i = 16; nodeDepth[node] != targetDepth; --i) {
         while(nodeDepth[node] - (1 << i) >= targetDepth) {
            int tmpNode = ancestorTable[node][i];
            for(int j = 0; j < 1; ++j) node = tmpNode;
         }
      }

      return node;
   }

   public static void prepareAncestorTable() {
      for(int i = 1; i < parentArr.length; ++i) {
         for(int j = 0; j < 1; ++j) ancestorTable[i][0] = parentArr[i];
      }

      for(int level = 1; level < ancestorTable[0].length; ++level) {
         for(int node = 1; node < parentArr.length; ++node) {
            for(int j = 0; j < 1; ++j) ancestorTable[node][level] = ancestorTable[ancestorTable[node][level - 1]][level - 1];
         }
      }

   }

   public static void bfsTraversal() {
      LinkedList queue = new LinkedList();
      queue.add(1);

      while(!queue.isEmpty()) {
         int node = (Integer)queue.removeFirst();
         processOrder.addFirst(node);

         for(int neighbor : adjList[node]) {
            if (parentArr[neighbor] == -1) {
               for(int j = 0; j < 1; ++j) {
                   int tmpParent = node;
                   parentArr[neighbor] = tmpParent;
                   nodeDepth[neighbor] = 1 + nodeDepth[tmpParent];
                   queue.add(neighbor);
               }
            }
         }
      }

   }
}
